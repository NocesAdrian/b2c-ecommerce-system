roles
    - id PK
    - name
    - description
    - date_created

users
    - id PK
    - role_id FK DEFAULT 2 COMMENT '1 = admin, 2 = customer'
    - username UNIQUE
    - password_hash
    - date_updated
    - date_created

sessions
    - id PK
    - user_id FK
    - token_hash UNIQUE
    - remember_me
    - expire_at
    - date_created

categories
    - id PK
    - name
    - date_created

products
    - id PK
    - category_id FK
    - name
    - description
    - unit_price CHECK (> 0)
    - stocks DEFAULT 0 CHECK (>= 0)
    - image_path
    - date_updated
    - date_created

carts
    - id PK
    - user_id FK UNIQUE
    - date_created

cart_items
    - cart_id FK
    - product_id FK
    - quantity CHECK (> 0)
    - date_updated
    - PK (cart_id, product_id)

saved_locations
    - id PK
    - user_id FK
    - location
    - date_created

orders
    - id PK
    - user_id FK
    - location_id FK
    - status DEFAULT 'pending'
        COMMENT 'pending, processing, shipped, delivered'
    - payment_method DEFAULT 'cash on delivery'
        COMMENT 'cash on delivery, online payment'
    - payment_status DEFAULT 'unpaid'
        COMMENT 'unpaid, paid'
    - total
    - date_created

order_items
    - order_id FK
    - product_id FK
    - quantity CHECK (> 0)
    - unit_price
    - PK (order_id, product_id)
